Fitur TikTok Downloader (Baileys / WhatsApp-web.js) 
const axios = require('axios');

async function tiktokHandler(sock, chatId, videoUrl) {
    try {
        const baseUrl = 'https://domain-anda.com/api/v1';
        const apiUrl = `${baseUrl}/download/ttdl?url=${encodeURIComponent(videoUrl)}`;

        const response = await axios.get(apiUrl);
        const res = response.data;

        if (res.status && res.result) {
            const videoLink = res.result.video || res.result.url;
            const title = res.result.title || 'TikTok Video';

            await sock.sendMessage(chatId, {
                video: { url: videoLink },
                caption: `✨ *TikTok Downloader*\n\n📌 *Judul:* ${title}`
            });
        } else {
            await sock.sendMessage(chatId, { text: '❌ Gagal mengambil video. Pastikan link publik!' });
        }
    } catch (err) {
        console.error(err);
        await sock.sendMessage(chatId, { text: '⚠️ Terjadi kesalahan pada server API.' });
    }
}
Fitur Telegram Bot (Python Telebot) 
import telebot
import requests

bot = telebot.TeleBot("YOUR_TELEGRAM_BOT_TOKEN")
BASE_URL = "https://domain-anda.com/api/v1"

@bot.message_handler(commands=['ytmp4'])
def download_yt(message):
    try:
        url = message.text.split(" ")[1]
        bot.reply_to(message, "⏳ Mengambil data video YouTube...")

        api_url = f"{BASE_URL}/download/ytmp4?url={url}&quality=720"
        res = requests.get(api_url).json()

        if res.get('status') and res.get('result'):
            video_url = res['result']['url']
            title = res['result'].get('title', 'YouTube Video')

            bot.send_video(message.chat.id, video_url, caption=f"🎬 {title}", parse_mode="HTML")
        else:
            bot.reply_to(message, "❌ Gagal memproses link YouTube.")
    except Exception as e:
        bot.reply_to(message, "⚠️ Gunakan format: /ytmp4 [LINK_YOUTUBE]")

Penggunaan di Node.js (Axios Client tanpa API Key) 
const axios = require('axios');

async function downloadSpotify(trackUrl) {
    const BASE_URL = 'https://domain-anda.com/api/v1';

    try {
        const response = await axios.get(`${BASE_URL}/download/spotifydl`, {
            params: { url: trackUrl }
        });

        console.log('Respon API:', response.data);
        return response.data.result;
    } catch (error) {
        console.error('Error fetching Spotify track:', error.message);
    }
}

Panggilan Client-Side (Fetch API) 
async function downloadInstagramMedia(igUrl) {
    const baseUrl = `${window.location.origin}/api/v1`;
    const apiUrl = `${baseUrl}/download/igdl?url=${encodeURIComponent(igUrl)}`;

    try {
        const response = await fetch(apiUrl);
        const json = await response.json();

        if (json.status) {
            console.log('Direct Link:', json.result.url);
            window.open(json.result.url, '_blank');
        } else {
            alert('Gagal mengambil data media!');
        }
    } catch (err) {
        console.error('Error:', err);
    }
}
